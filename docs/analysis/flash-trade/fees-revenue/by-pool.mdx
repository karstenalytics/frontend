export const poolFilter = {
  _pools: null,
  _cbs: new Set(),
  _colors: {},
  get() { return this._pools; },
  set(v) { this._pools = v; this._cbs.forEach(fn => fn(v)); },
  sub(fn) { this._cbs.add(fn); return () => this._cbs.delete(fn); },
  getColors() { return this._colors; },
  setColors(m) { this._colors = m; },

# Fees & Revenue by Pool

Fees and revenue categorized by liquidity pool over time. Toggle between daily and cumulative views, and switch between fee and revenue metrics.

## Data

<BrowserOnly fallback={<div style={{ minHeight: '800px', display: 'flex', alignItems: 'center', justifyContent: 'center' }}><LoadingSpinner /></div>}>
  {() => {
    const { useState, useEffect } = require('react');
    const PoolModule = require('@site/src/components/FlashTrade/PoolStackedAreaChart');
    const PoolStackedAreaChart = PoolModule.default;

    function ChartSection() {
      const [visiblePools, setLocalPools] = useState(poolFilter.get());

      useEffect(() => {
        poolFilter._cbs.add(setLocalPools);
        return () => poolFilter._cbs.delete(setLocalPools);
      }, []);

      const setVisiblePools = (pools) => {
        poolFilter.set(pools);
      };

      return (
        <PoolStackedAreaChart
          visiblePools={visiblePools}
          onVisibilityChange={setVisiblePools}
          onColorsComputed={(colors) => poolFilter.setColors(colors)}
        />
      );
    }

    return <ChartSection />;
  }}
</BrowserOnly>

## Top Transactions

The table displays the largest individual transactions contributing to protocol fees. Clicking a pool in the chart or legend filters this table to that pool's transactions.

<BrowserOnly fallback={<div style={{ minHeight: '200px', display: 'flex', alignItems: 'center', justifyContent: 'center' }}><LoadingSpinner /></div>}>
  {() => {
    const { useState, useEffect } = require('react');
    const FilterIndicator = require('@site/src/components/common/FilterIndicator').default;
    const { tableStyles, tableRowHoverHandlers, linkHoverHandlers } = require('@site/src/styles/tableStyles');

    function TransactionsTable() {
      const [topTxData, setTopTxData] = useState(null);
      const [loading, setLoading] = useState(true);
      const [error, setError] = useState(null);
      const [visiblePools, setLocalPools] = useState(poolFilter.get());

      useEffect(() => {
        poolFilter._cbs.add(setLocalPools);
        return () => poolFilter._cbs.delete(setLocalPools);
      }, []);

      useEffect(() => {
        fetch('/data/flash-trade/top_transactions_pool.json')
          .then(r => r.ok ? r.json() : null)
          .then(topTx => {
            setTopTxData(topTx);
            setLoading(false);
          })
          .catch(err => {
            setError(err.message);
            setLoading(false);
          });
      }, []);

      const consolidatePoolName = (name) => name.replace(/ \(V[0-9](?:\/V[0-9])?\)$/, '');

      const clearFilter = () => {
        poolFilter.set(null);
      };

      if (loading) {
        return (
          <div style={{ minHeight: '200px', display: 'flex', alignItems: 'center', justifyContent: 'center' }}>
            <LoadingSpinner />
          </div>
        );
      }

      if (error) {
        return (
          <div style={{
            padding: '32px',
            background: 'var(--ifm-background-surface-color)',
            border: '1px solid var(--ifm-color-danger)',
            borderRadius: 'var(--ifm-global-radius)',
            color: 'var(--ifm-color-danger)',
          }}>
            <div style={{ fontSize: '1.25rem', fontWeight: 600, marginBottom: '8px' }}>Error Loading Data</div>
            <p>{error}</p>
          </div>
        );
      }

      const allTransactions = topTxData?.transactions || [];
      const filteredTransactions = visiblePools
        ? allTransactions.filter(tx => visiblePools.includes(consolidatePoolName(tx.pool)))
        : allTransactions;
      const displayTransactions = filteredTransactions.slice(0, 10);

      const colors = poolFilter.getColors();
      const filterChips = visiblePools
        ? visiblePools.map(pool => ({ label: pool, color: colors[pool] || '#888888' }))
        : [];

      return (
        <div>
          <FilterIndicator filters={filterChips} onClear={clearFilter} />

          {topTxData ? (
            <div style={{ overflowX: 'auto' }}>
              <table style={tableStyles.table}>
                <thead>
                  <tr style={tableStyles.headerRow}>
                    <th style={tableStyles.headerCell}>#</th>
                    <th style={tableStyles.headerCell}>Date</th>
                    <th style={tableStyles.headerCell}>Pool</th>
                    <th style={tableStyles.headerCell}>Type</th>
                    <th style={{ ...tableStyles.headerCell, textAlign: 'right' }}>Fee (USDC)</th>
                    <th style={tableStyles.headerCell}>Transaction</th>
                  </tr>
                </thead>
                <tbody>
                  {displayTransactions.length > 0 ? displayTransactions.map((tx, idx) => {
                    const poolName = consolidatePoolName(tx.pool);
                    const color = colors[poolName] || '#888888';
                    return (
                      <tr key={tx.signature} style={tableStyles.bodyRow} {...tableRowHoverHandlers}>
                        <td style={{ ...tableStyles.cell, fontWeight: 600 }}>{idx + 1}</td>
                        <td style={tableStyles.dateCell}>{tx.date}</td>
                        <td style={tableStyles.cell}>
                          <span style={{
                            display: 'inline-block',
                            width: '10px',
                            height: '10px',
                            borderRadius: '2px',
                            backgroundColor: color,
                            marginRight: '6px',
                          }} />
                          {poolName}
                        </td>
                        <td style={tableStyles.cell}>{tx.type}</td>
                        <td style={{ ...tableStyles.cell, textAlign: 'right', fontWeight: 600 }}>
                          ${tx.fee_usdc.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}
                        </td>
                        <td style={tableStyles.addressCell}>
                          <a
                            href={`https://solscan.io/tx/${tx.signature}`}
                            target="_blank"
                            rel="noopener noreferrer"
                            style={tableStyles.link}
                            {...linkHoverHandlers}
                          >
                            {tx.signature.slice(0, 8)}...{tx.signature.slice(-8)}
                          </a>
                        </td>
                      </tr>
                    );
                  }) : (
                    <tr>
                      <td colSpan={6} style={{ ...tableStyles.cell, textAlign: 'center', padding: '24px' }}>
                        {visiblePools ? 'No transactions found for selected pools' : 'No transactions available'}
                      </td>
                    </tr>
                  )}
                </tbody>
              </table>
            </div>
          ) : (
            <div style={{
              padding: '32px',
              textAlign: 'center',
              background: 'var(--ifm-background-surface-color)',
              border: '1px solid var(--ifm-toc-border-color)',
              borderRadius: 'var(--ifm-global-radius)',
              color: 'var(--ifm-color-secondary)',
            }}>
              Transaction data not yet available. The data pipeline will generate this file on the next run.
            </div>
          )}
        </div>
      );
    }

    return <TransactionsTable />;
  }}
</BrowserOnly>

## Use Cases

This chart helps you:

- **Track fee and revenue trends**: Monitor how each pool's contribution changes over time
- **Explore fee composition**: Use the metric toggles to compare Fees (trading + non-trading) and Revenue (protocol fee distributions)
- **Identify top performers**: See which perpetual markets generate the most protocol fees
- **Compare fee magnitudes**: Switch metrics to see how gross fees compare to protocol revenue distributions
- **Assess pool concentration**: Determine if revenue is concentrated in specific markets or distributed
- **Investigate specific pools**: Click a pool's legend item or chart bar/area to filter the transactions table and examine individual trades

  - Use the metric toggle to switch between Fees and Revenue

  - Double-click a legend item to isolate a single pool
  - Hover to see values per pool

- **Date**: When the transaction occurred (UTC)
- **Pool**: Which market generated the fee
- **Type**: Trading operation type (CloseAndSwap, OpenPosition, Liquidate, etc.)

## Methodology

<details>
<summary>Click to expand technical details</summary>

**Data Sources**

- **Treasury transactions**: On-chain data from the Flash.Trade stake pool authority
- **Fee attribution**: Internal fee events decoded from transaction logs
- **Pool identification**: Borsh-encoded pool identifiers in CPI event data

**Pool Types**

Flash.Trade operates multiple perpetual markets (pools):

| Pool | Market | Description |
|------|--------|-------------|
| Crypto.1 | BTC/ETH/SOL | Main crypto perpetuals (largest volume) |
| Governance.1 | Governance tokens | Governance token perpetuals |
| Community.1 | Community tokens | Community/meme token perpetuals |
| Community.2 | Community tokens | Additional community markets |
| Community.3 | Community tokens | Additional community markets (last active July 2025) |
| Trump.1 | TRUMP | TRUMP memecoin perpetuals |
| Ore.1 | ORE | ORE token perpetuals |
| Virtual.1 | Virtual tokens | Virtual token perpetuals |
| Remora.1 | Remora tokens | Remora perpetuals |

**Pool Version Consolidation**

The backend tracks contract versions separately (e.g., "Crypto.1 (V1/V2)", "Crypto.1 (V3)"). The data pipeline strips all version suffixes (V1, V2, V3, V1/V2) and consolidates by base name before delivery, as the version distinction is an implementation detail not relevant for revenue analysis.

**Fee Attribution**

Each trading operation generates fees that are attributed to a specific pool:
- Parse internal fee events from transaction logs
- Decode Borsh-encoded pool identifier
- Map to human-readable pool name
- Aggregate by date and pool

**Update Frequency**

Data is automatically updated once per day. Check the "Last updated" timestamp in the site header for data freshness.

**Known Limitations**

- Transaction table stores up to 10 entries per pool and displays the top 10 for the active view; applying a pool filter reveals that pool's individual top 10
- Pool versions are consolidated (V1/V2 shown together)
- New pools may appear with generated names until manually labeled
- Revenue metric shows zero for April 12 - June 18, 2025 because MoveProtocolFees tracking began on June 19, 2025
- Fees metric shows trading + non-trading fees; LP fees are excluded (less than 1% of total)

</details>
